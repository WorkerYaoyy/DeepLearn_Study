# Shuzishibie.py 代码分析 & 优化方案

---

## 一、代码结构与执行流程

```
导入依赖 → 模型定义(784→256→128→10)
    → 数据准备(MNIST下载 + 展平归一化)
    → 训练(5轮Adam) → 保存 mnist_model.pth
    → 取测试集图片 → 预测 → 可视化
```

### 流程特点

| 项目 | 现状 |
|------|------|
| 每次执行都训练？ | ✅ **是的，每次都从头训练** |
| 训练完保存模型？ | ✅ 保存为 `mnist_model.pth` |
| 启动时检查已有模型？ | ❌ **不检查，直接重新训练** |
| 浪费情况 | 若已训练过，每次运行浪费约 30~60 秒 GPU/CPU 时间 |

## 二、核心问题：重复训练

```python
# 第74行：永远创建新模型（随机初始化参数）
model = MnistModel()

# 第79-99行：无条件执行训练循环
for epoch in range(epochs):
    ...

# 第101行：训练完才保存
torch.save(model.state_dict(), "mnist_model.pth")
```

**问题本质**：第74行和第79行之间没有任何判断逻辑。即使上次运行已经训练好并保存了 `mnist_model.pth`，下次运行也会**忽略它**，重新随机初始化、重新训练。

## 三、优化方案：缓存判断机制

### 核心思路

```
程序启动
    │
    ├── mnist_model.pth 存在？
    │       │
    │       ├── 是 → 直接加载 → 跳过训练
    │       │
    │       └── 否 → 训练 → 保存 → 继续
    │
    └── 预测 & 可视化
```

### 优化后完整代码

```python
"""
MNIST 手写数字识别 —— 三层 MLP 完整流程（带模型缓存）
结构: 784 → 256 → ReLU → 128 → ReLU → 10
"""

# ============================================================
# 0. 环境配置
# ============================================================
import os
import ssl

ssl._create_default_https_context = ssl._create_unverified_context

import torch
import torch.nn as nn
from torch import Tensor
from torch.utils.data import DataLoader
import torchvision
import torchvision.transforms as transforms
import matplotlib.pyplot as plt


# ============================================================
# 1. 模型定义
# ============================================================
class MnistModel(nn.Module):
    """三层全连接网络: 784 → 256 → ReLU → 128 → ReLU → 10"""

    def __init__(self) -> None:
        super().__init__()
        self.layer1 = nn.Linear(784, 256)
        self.relu1 = nn.ReLU()
        self.layer2 = nn.Linear(256, 128)
        self.relu2 = nn.ReLU()
        self.layer3 = nn.Linear(128, 10)

    def forward(self, x: Tensor) -> Tensor:
        x = self.relu1(self.layer1(x))
        x = self.relu2(self.layer2(x))
        x = self.layer3(x)
        return x


# ============================================================
# 2. 数据准备 —— transform 统一完成 展平 + 归一化
# ============================================================
def img_preprocess(img) -> Tensor:
    """PIL Image → 展平成一维 → 归一化到 [0, 1]"""
    tensor = transforms.functional.to_tensor(img)  # 自动转 float 并 /255
    return tensor.view(28 * 28)


train_dataset = torchvision.datasets.MNIST(
    root="./dataset",
    train=True,
    download=True,
    transform=img_preprocess,
)
test_dataset = torchvision.datasets.MNIST(
    root="./dataset",
    train=False,
    download=True,
    transform=img_preprocess,
)

train_loader = DataLoader(train_dataset, batch_size=50, shuffle=True)
test_loader = DataLoader(test_dataset, batch_size=50, shuffle=False)

# ============================================================
# 3. 模型加载 / 训练（核心：缓存判断）
# ============================================================
MODEL_PATH = "Training_dataset/mnist_model.pth"
model = MnistModel()

if os.path.exists(MODEL_PATH):
    # --- 已有模型：直接加载 ---
    model.load_state_dict(torch.load(MODEL_PATH, weights_only=True))
    print(f"✅ 已加载已有模型 → {MODEL_PATH}，跳过训练")
else:
    # --- 无模型：从头训练 ---
    print("⚠️ 未找到模型文件，开始训练...")
    criterion = nn.CrossEntropyLoss()
    optimizer = torch.optim.Adam(model.parameters(), lr=0.001)
    epochs = 5

    for epoch in range(epochs):
        model.train()
        total_loss = 0.0
        correct = 0
        total = 0

        for images, labels in train_loader:
            logits = model(images)
            loss = criterion(logits, labels)

            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            total_loss += loss.item() * images.size(0)
            correct += (logits.argmax(dim=1) == labels).sum().item()
            total += images.size(0)

        avg_loss = total_loss / total
        acc = correct / total
        print(f"  Epoch {epoch + 1}/{epochs} — loss: {avg_loss:.4f}, acc: {acc:.4f}")

    torch.save(model.state_dict(), MODEL_PATH)
    print(f"✅ 训练完成，模型已保存 → {MODEL_PATH}")

# ============================================================
# 4. 预测与可视化
# ============================================================
# 取测试集中某张图片（可自行修改索引）
img_idx = 88
img, label = test_dataset[img_idx]

model.eval()
with torch.no_grad():
    logits = model(img.unsqueeze(0))  # [784] → [1, 784]
    probs = torch.softmax(logits, dim=1)
    pred = probs.argmax(dim=1).item()
    confidence = probs[0, pred].item()

print(f"图片索引: {img_idx}")
print(f"真实标签: {label},  预测结果: {pred},  置信度: {confidence:.4f}")

plt.imshow(img.view(28, 28), cmap="gray")
plt.title(f"True: {label}  |  Pred: {pred}  ({confidence:.1%})")
plt.axis("off")
plt.show()
```

## 六、扩展：输入 PNG 图片进行识别

### 整体流程

```
my_digit.png → PIL 打开 → 灰度化 → resize(28×28)
    → 检查是否需要反转颜色 → to_tensor() → [0,1]
    → view(784) → 送入模型 → softmax → 预测结果
```

### 核心代码（添加到 Shuzishibie.py 第4部分之后）

```python
# ============================================================
# 5. 单张 PNG 图片识别
# ============================================================
from PIL import Image

def predict_image(model, image_path: str):
    """
    输入一张 PNG 图片，返回预测数字和置信度
    
    参数:
        model: 训练好的 MnistModel
        image_path: PNG 图片路径
    
    返回:
        (预测数字, 置信度)
    """
    model.eval()

    # 1. 读取图片 → 灰度 → 28×28
    img = Image.open(image_path).convert("L")
    img = img.resize((28, 28), Image.LANCZOS)

    # 2. 判断是否需要反转颜色（见下方说明）
    #   MNIST 格式: 黑底(0.0) 白字(1.0)
    #   如果你的图是 白底黑字 → 需要反转
    tensor = transforms.functional.to_tensor(img)    # → [1, 28, 28], 值域 [0,1]

    # ★ 取消下面这行的注释即可反转颜色 ★
    # tensor = 1.0 - tensor

    # 3. 推理
    with torch.no_grad():
        logits = model(tensor.view(1, 784))          # [1, 784]
        probs = torch.softmax(logits, dim=1)
        pred = probs.argmax(dim=1).item()
        confidence = probs[0, pred].item()

    return pred, confidence


# --- 使用示例 ---
img_path = "my_digit.png"            # 改成你的图片路径
pred, conf = predict_image(model, img_path)
print(f"预测结果: {pred}  (置信度: {conf:.1%})")

# 显示图片
img = Image.open(img_path).convert("L").resize((28, 28), Image.LANCZOS)
plt.imshow(img, cmap="gray")
plt.title(f"Pred: {pred}  ({conf:.1%})")
plt.axis("off")
plt.show()
```

### ⚠️ 关键：颜色方向（最常见的坑）

MNIST 数据格式是 **黑底白字**（笔画=亮=1.0，背景=暗=0.0）。

你的 PNG 图片有两种情况：

```
【情况A】黑底白字 — 无需反转          【情况B】白底黑字 — 需要反转
  ████████████                         ··············
  ████····████                         ····██████····
  ███······███                         ····██████····
  ███······███          →              ····██████····
  ████····████                         ····██████····
  ████████████                         ··············
  (和MNIST一致 ✅)                      (和MNIST相反 ❌ → tensor = 1.0 - tensor)
```

**判断方法**：查看图片背景色
- 背景是黑色 → 不需要反转
- 背景是白色 → 取消 `tensor = 1.0 - tensor` 的注释

### 完整数据流对照

```
【训练时 MNIST】                   【推理时 PNG 图片】
PIL Image (28×28, 灰度)            my_digit.png → convert("L")
    │                                   │
    v                                   v
to_tensor() → [0,1]              resize(28×28) → 颜色检查
    │                                   │
    v                                   v
view(784)                        to_tensor() → 反转?(视情况)
    │                                   │
    └─────────── 相同格式 ─────────────┘
                    │
                    v
              MnistModel → softmax → 数字
```

### 可以用什么画 PNG？

| 工具 | 操作 |

### 优化建议

```python
def predict_image_robust(model, image_path: str):
    """
    自动判断颜色方向：哪种方向让模型更"确定"，就用哪种
    适用于不确定图片是黑底还是白底的情况
    """
    model.eval()
    img = Image.open(image_path).convert("L").resize((28, 28), Image.LANCZOS)
    tensor = transforms.functional.to_tensor(img).view(1, 784)

    with torch.no_grad():
        # 分别试两种颜色方向
        probs_normal = torch.softmax(model(tensor), dim=1)        # 不反转
        probs_invert = torch.softmax(model(1.0 - tensor), dim=1)  # 反转

        # 取置信度更高的那个
        conf_normal = probs_normal.max().item()
        conf_invert = probs_invert.max().item()

        if conf_normal >= conf_invert:
            pred = probs_normal.argmax(dim=1).item()
            return pred, conf_normal, "正常(不反转)"
        else:
            pred = probs_invert.argmax(dim=1).item()
            return pred, conf_invert, "反转"


# 使用
pred, conf, mode = predict_image_robust(model, "my_digit.png")
print(f"预测: {pred}, 置信度: {conf:.1%}, 模式: {mode}")
```
